{# ============================= API MACHINE ============================== #}

{% translate "API machine (HMAC)" %}

{% translate "Surface Zero Trust consommée par le SDK et un éventuel LMS : aucune session Django, aucun JWT. Le serveur signe une chaîne canonique avec sa clé secrète. ExamSDK ne fait confiance à rien d'autre que cette signature." %}

{# --- Signer une requête --- #}

{% translate "Signer une requête" %}

{% translate "Chaque requête porte deux en-têtes :" %}

  • X-ExamSDK-Public-Key — {% translate "la clé publique du projet (jamais secrète)." %}
  • X-ExamSDK-Signature — {% translate "le HMAC-SHA256 hexadécimal de la chaîne canonique, calculé avec la clé secrète." %}
{% translate "Endpoint" %}{% translate "Chaîne canonique" %}
sessions /project_id|exam_id|student_id|issued_at
sessions /close/project_id|session_id|issued_at
events /project_id|session_id|issued_at

{% translate "issued_at est un horodatage epoch (secondes) ; il doit être frais (fenêtre de tolérance serveur). Pour les appels GET, les champs signés arrivent par les query params." %}

{% translate "Signer en Python" %}

{% code_block "doc-machine-sign-python" "python" "sign.py" %} import hashlib import hmac import time SECRET_KEY = "sk_live_..." # clé secrète du projet (jamais exposée au navigateur) PROJECT_ID = "3c8e1f2a-9b47-4d0e-8a11-2f6c9d4b7e01" def sign(canonical: str) -> str: return hmac.new(SECRET_KEY.encode(), canonical.encode(), hashlib.sha256).hexdigest() issued_at = int(time.time()) canonical = "|".join([PROJECT_ID, "MATH-2024-S1", "etu-1042", str(issued_at)]) signature = sign(canonical) {% endcode_block %}
{# --- Ouvrir une session --- #}
POST /api/v1/machine/sessions/

{% translate "Ouvrir une session d'examen" %}

{% translate "Ouvre une session pour un couple (examen, étudiant) et renvoie son identifiant. L'appel est idempotent : si une session ACTIVE identique existe déjà, elle est renvoyée telle quelle (200) au lieu d'en créer une seconde (201)." %}

{% code_block "doc-machine-session-create-curl" "bash" "cURL" %} curl -X POST https://api.examsdk.example/api/v1/machine/sessions/ \ -H "X-ExamSDK-Public-Key: pk_live_9f2c7a1b4e..." \ -H "X-ExamSDK-Signature: " \ -H "Content-Type: application/json" \ -d '{"exam_id": "MATH-2024-S1", "student_id": "etu-1042", "issued_at": 1729000000, "sdk_version": "1.2.0"}' {% endcode_block %}

{% translate "Réponse" %} 201 {% translate "ou" %} 200
